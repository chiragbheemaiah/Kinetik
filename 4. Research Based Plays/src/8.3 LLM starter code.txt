2.1 models.py ? Pydantic models mirroring schemas
from __future__ import annotations

from typing import List, Optional
from pydantic import BaseModel, Field


class PriorityItem(BaseModel):
    id: str
    title: str
    summary: str


class SimpleItem(BaseModel):
    id: str
    summary: str


class OrgProfileSummary(BaseModel):
    account_id: str
    strategic_priorities: List[PriorityItem]
    it_initiatives: List[PriorityItem]
    key_pain_points: List[SimpleItem]
    regulatory_drivers: List[SimpleItem]


class AxisDefinition(BaseModel):
    axis_id: str
    label: str
    description: str
    weight: float = Field(ge=0, le=1)


class AxisScore(BaseModel):
    axis_id: str
    score: float = Field(ge=0, le=1)
    rationale: str


class OperatingUnitLink(BaseModel):
    operating_unit_id: str
    relationship_rationale: str


class PriorityLink(BaseModel):
    priority_id: str
    relationship_rationale: str


class UseCaseAssessment(BaseModel):
    tier2_id: str
    tier2_name: str
    tier1_id: str
    tier1_name: str
    axis_scores: List[AxisScore]
    overall_impact: float = Field(ge=0, le=1)
    overall_rationale: str
    linked_operating_units: List[OperatingUnitLink]
    linked_org_priorities: List[PriorityLink]


class UseCaseRankingOutput(BaseModel):
    account_id: str
    top_n_requested: int = Field(ge=1)
    axis_definitions: List[AxisDefinition]
    considered_usecases: int = Field(ge=0)
    usecases_ranked: List[UseCaseAssessment]


class RankingIssue(BaseModel):
    severity: str  # "info" | "warning" | "error"
    issue_code: str
    message: str
    related_usecase_id: Optional[str]
    suggested_action: str


class RankingEvaluationReport(BaseModel):
    account_id: str
    summary: str
    issues: List[RankingIssue]

### 2.3 `openai_client.py` ? Thin wrapper around Responses API

```python
import json
from typing import Any, Dict

from openai import OpenAI

from .schemas import ORG_PROFILE_SUMMARY_SCHEMA, USECASE_RANKING_SCHEMA
from . import prompts

client = OpenAI()


def _call_structured_json(
    *,
    model: str,
    system_prompt: str,
    developer_instructions: str,
    user_input: str,
    schema_name: str,
    schema: Dict[str, Any],
    temperature: float = 0.2,
    reasoning_effort: str | None = None,
) -> Dict[str, Any]:
    """
    Helper: call OpenAI Responses API with structured JSON output.
    """
    params: Dict[str, Any] = {
        "model": model,
        "input": [
            {"role": "system", "content": system_prompt},
            {"role": "developer", "content": developer_instructions},
            {"role": "user", "content": user_input},
        ],
        "response_format": {
            "type": "json_schema",
            "json_schema": {
                "name": schema_name,
                "schema": schema,
                "strict": True,
            },
        },
        "temperature": temperature,
    }
    if reasoning_effort is not None:
        params["reasoning"] = {"effort": reasoning_effort}

    resp = client.responses.create(**params)
    text = resp.output[0].content[0].text
    return json.loads(text)


def summarize_org_profile(
    *,
    model: str,
    account_name: str,
    account_id: str,
    org_profile_json_str: str,
) -> Dict[str, Any]:
    user_input = prompts.USER_STAGE_A_TEMPLATE.format(
        account_name=account_name,
        account_id=account_id,
        org_profile_json=org_profile_json_str,
    )
    return _call_structured_json(
        model=model,
        system_prompt=prompts.SYSTEM_STAGE_A,
        developer_instructions=prompts.DEVELOPER_STAGE_A,
        user_input=user_input,
        schema_name="OrgProfileSummary",
        schema=ORG_PROFILE_SUMMARY_SCHEMA,
    )


def rank_usecases(
    *,
    model: str,
    reasoning_effort: str,
    account_name: str,
    account_id: str,
    top_n: int,
    summarized_profile_json_str: str,
    operating_units_json_str: str,
    taxonomy_json_str: str,
    scoring_axes_json_str: str,
) -> Dict[str, Any]:
    developer_instructions = prompts.DEVELOPER_STAGE_B_TEMPLATE.format(
        scoring_axes_json=scoring_axes_json_str
    )
    user_input = prompts.USER_STAGE_B_TEMPLATE.format(
        account_name=account_name,
        account_id=account_id,
        top_n=top_n,
        summarized_profile_json=summarized_profile_json_str,
        operating_units_json=operating_units_json_str,
        taxonomy_json=taxonomy_json_str,
    )
    return _call_structured_json(
        model=model,
        system_prompt=prompts.SYSTEM_STAGE_B,
        developer_instructions=developer_instructions,
        user_input=user_input,
        schema_name="UseCaseRankingOutput",
        schema=USECASE_RANKING_SCHEMA,
        reasoning_effort=reasoning_effort,
    )

2.4 chapter_8_3_usecase_selector.py ? CLI entrypoint on AWS
This is the script you actually run on the EC2 box.
"""
Chapter 8.3 ? Top Use Cases per Account

CLI entrypoint that:
1) Loads org profile + operating units + taxonomy.
2) Uses OpenAI models to summarize the org profile (Stage A).
3) Uses OpenAI models to score and rank use cases (Stage B).
4) Writes JSON outputs for downstream consumption.
"""

import argparse
import json
from pathlib import Path
from typing import Any, Dict

import yaml

from . import openai_client
from .models import OrgProfileSummary, UseCaseRankingOutput


def load_yaml(path: Path) -> Dict[str, Any]:
    with path.open("r", encoding="utf-8") as f:
        return yaml.safe_load(f)


def load_json(path: Path) -> Any:
    with path.open("r", encoding="utf-8") as f:
        return json.load(f)

def main() -> None:
    parser = argparse.ArgumentParser(description="Chapter 8.3 ? Top Use Cases per Account")
    parser.add_argument("--account-id", required=True, help="Account ID, e.g. CITI or STATE_OF_TEXAS.")
    parser.add_argument("--account-name", required=True, help="Humanreadable account name.")
    parser.add_argument("--config", default="config.yaml", help="Path to chapter 8.3 config YAML.")
    args = parser.parse_args()

    base_dir = Path(__file__).resolve().parent
    chapter_root = base_dir
    cfg = load_yaml(Path(args.config))

    account_id = args.account_id
    account_name = args.account_name

    # Paths ? adjust to your repo layout
    org_profile_path = chapter_root / "inputs" / "org_profiles" / f"{account_id}_org_profile.json"
    operating_units_path = Path("..") / "chapter_6" / "outputs" / f"{account_id}_operating_units.json"
    taxonomy_path = chapter_root / "data" / "taxonomy_tiers_definitions.xlsx"

    org_profile = load_json(org_profile_path)
    operating_units = load_json(operating_units_path)
    taxonomy = load_taxonomy(taxonomy_path)

    # Convert to JSON strings for prompts
    org_profile_json_str = json.dumps(org_profile, ensure_ascii=False, indent=2)
    operating_units_json_str = json.dumps(operating_units, ensure_ascii=False, indent=2)
    taxonomy_json_str = json.dumps(taxonomy, ensure_ascii=False, indent=2)
    scoring_axes_json_str = json.dumps(cfg["scoring_axes"], ensure_ascii=False, indent=2)

    # Stage A ? org profile summarization
    summary_raw = openai_client.summarize_org_profile(
        model=cfg.get("stage_a_model", "gpt-5-mini"),   # or "gpt-4.1-mini"
        account_name=account_name,
        account_id=account_id,
        org_profile_json_str=org_profile_json_str,
    )
    summary = OrgProfileSummary.model_validate(summary_raw)
    summarized_profile_json_str = summary.model_dump_json(indent=2, ensure_ascii=False)

    # TODO (optional): filter taxonomy by industry/segment before sending to LLM
    taxonomy_subset_json_str = taxonomy_json_str

    # Stage B ? usecase scoring and ranking
    ranking_raw = openai_client.rank_usecases(
        model=cfg.get("stage_b_model", "gpt-5.1"),
        reasoning_effort=cfg.get("stage_b_reasoning_effort", "medium"),  # "low" | "medium" | "high"
        account_name=account_name,
        account_id=account_id,
        top_n=cfg.get("top_n_usecases", 10),
        summarized_profile_json_str=summarized_profile_json_str,
        operating_units_json_str=operating_units_json_str,
        taxonomy_json_str=taxonomy_subset_json_str,
        scoring_axes_json_str=scoring_axes_json_str,
    )
    ranking = UseCaseRankingOutput.model_validate(ranking_raw)

    # TODO: validate overall_impact vs axis_scores and log warnings.

    outputs_dir = chapter_root / "outputs"
    outputs_dir.mkdir(parents=True, exist_ok=True)
    output_path = outputs_dir / f"{account_id}_8_3_usecases.json"

    with output_path.open("w", encoding="utf-8") as f:
        json.dump(ranking.model_dump(), f, ensure_ascii=False, indent=2)

    print(f"Wrote 8.3 ranking to {output_path}")


if __name__ == "__main__":
    main()
